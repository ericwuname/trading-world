A.3.1 先修离散化偏差(H3的前置修正,必须先做,否则后续实验的"平均活跃度不变"这个前提不成立)

# tw/order_flow/hawkes.py 修改

class HawkesIntensityProcess:
    def __init__(self, mu, alpha, beta, dt=1.0):
        assert alpha < beta * (np.exp(beta * dt) - 1) / dt if dt > 0 else alpha < beta
        self.mu, self.alpha, self.beta, self.dt = mu, alpha, beta, dt

    @classmethod
    def from_target_mean_rate(cls, target_lambda_bar, alpha, beta, dt=1.0):
        """
        报告发现的偏差来源:连续公式的平稳强度是 mu/(1-alpha/beta),
        但离散递推下真正的分支比是 alpha/(exp(beta*dt)-1),不是alpha/beta。
        用正确的离散分支比反解mu,确保实际平均到达率精确命中target_lambda_bar。
        """
        discrete_branching_ratio = alpha / (np.exp(beta * dt) - 1)
        assert discrete_branching_ratio < 1, "离散分支比>=1,强度过程不稳定"
        mu = target_lambda_bar * (1 - discrete_branching_ratio)
        return cls(mu, alpha, beta, dt)

A.3.3 突发期/背景期深度对比工具（检验H1）

# tw/analyzer_impact.py

def classify_burst_vs_background(hawkes_intensity_trace, percentile_threshold=80):
    """
    对每个tick,用当时的Hawkes强度值,标记是否处于'突发期'
    (强度超过该次模拟运行的第80百分位)。
    """
    threshold = np.percentile(hawkes_intensity_trace, percentile_threshold)
    return hawkes_intensity_trace > threshold

def compare_depth_burst_vs_background(depth_snapshots, burst_mask):
    """
    返回突发期与背景期的平均盘口深度,以及配对t检验
    (这里的'配对'是同一次模拟运行内的两组样本比较,不是跨模拟的种子配对,
    要在报告里明确写清楚是哪一种'配对'避免混淆一期/二期反复强调的'同种子配对'术语)。
    """
    burst_depth = depth_snapshots[burst_mask]
    background_depth = depth_snapshots[~burst_mask]
    t, p = scipy.stats.ttest_ind(burst_depth, background_depth)
    return {"burst_mean": burst_depth.mean(), "background_mean": background_depth.mean(),
            "t": t, "p": p, "pct_diff": (burst_depth.mean()-background_depth.mean())/background_depth.mean()}

class MetaOrderZeroIntelligence(MetaOrderMixin, ZeroIntelligence):
    """
    零智能agent原本每次决策独立随机选方向。元订单化后:
    以meta_config.p_meta_start的概率启动一个'方向持续一段时间'的元订单,
    这段时间内后续的随机决策方向被元订单的方向覆盖(不再重新随机选边)。
    代表真实世界里'大额但无信息的流动性需求'(比如大户提现、清算债务),
    区别于图表派的'因为看到趋势才持续买'——这里没有任何方向判断依据,
    纯粹是外生需要执行一笔大单。
    """
    def on_tick(self, market_state):
        child = self.get_child_order_qty()
        if child is not None:
            direction, qty = child
            return self.buy_order(qty) if direction > 0 else self.sell_order(qty)
        if self._rng.random() < self.p_active:
            self.maybe_start_meta_order(base_direction_signal=self._rng.choice([1, -1]))
        return super().on_tick(market_state)

# tw/agents/fundamentalist.py 修改

class MetaOrderFundamentalist(MetaOrderMixin, Fundamentalist):
    """
    基本面派原本的方向由premium决定,力度连续。元订单化后:
    当|premium|超过某个阈值,不是马上把'全部想要的调仓量'一次性挂出去,
    而是启动一个元订单,分批执行,方向锚定在'元订单启动那一刻'的premium方向,
    即使执行过程中premium已经部分修正也不改变方向(直到元订单执行完)。
    这代表真实世界机构基于估值判断的大宗建仓/减仓行为——
    这类订单流的长记忆应该带有'均值回归'的味道(因为premium本身是均值回归的),
    区别于图表派带来的'趋势强化'味道的长记忆。这是B工作线要特别关注的一个定性差异。
    """
    def on_tick(self, market_state):
        child = self.get_child_order_qty()
        if child is not None:
            direction, qty = child
            return self.buy_order(qty) if direction > 0 else self.sell_order(qty)
        premium = market_state.premium()
        if abs(premium) > self.deadband:
            direction = -1 if premium > 0 else 1
            self.maybe_start_meta_order(base_direction_signal=direction)
        return None
