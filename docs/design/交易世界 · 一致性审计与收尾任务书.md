def ci_overlap_fraction(ci_a: tuple, ci_b: tuple) -> float:
    """
    复用EL.2里已经用过的重叠度量逻辑,抽成独立可测试的函数
    (EL.2当时可能是内联算的,这次要求抽出来,避免以后每次又重新写一遍类似逻辑)。
    返回:两个区间的重叠长度 / 较窄那个区间的宽度。
    """
    lo = max(ci_a[0], ci_b[0])
    hi = min(ci_a[1], ci_b[1])
    overlap = max(0.0, hi - lo)
    narrower_width = min(ci_a[1] - ci_a[0], ci_b[1] - ci_b[0])
    return overlap / narrower_width if narrower_width > 0 else np.nan

def compute_direction_verdict(k_point, ci):
    """
    新的验收标准:不再要求'k要接近0.5'这个精确目标,
    改成'CI是否排除0.5(即是否能确定显著偏离平方根律)'这个方向性判据。
    三个输出:
    - 'sqrt_law_excluded': CI不包含0.5,即确定偏离平方根律(需要进一步看点估计
      在0.5的哪一侧,是偏线性还是偏更凹)
    - 'sqrt_law_consistent': CI包含0.5但不包含1.0,即不能排除接近平方根律
    - 'undetermined': CI同时包含0.5和1.0(或者其他更复杂的情况),完全无法判断
    """
    excludes_0_5 = not (ci[0] <= 0.5 <= ci[1])
    excludes_1_0 = not (ci[0] <= 1.0 <= ci[1])
    if excludes_0_5:
        direction = "偏线性(k>0.5)" if k_point > 0.5 else "偏超凹(k<0.5)"
        return f"sqrt_law_excluded_{direction}"
    elif excludes_1_0:
        return "sqrt_law_consistent"
    else:
        return "undetermined"

工作线O的结论是本轮最重要的产出，优先看这个。如果EO.2/EO.4确认此前 报的"反转"其实也是噪声，这意味着本轮报告里"两条历史结论被推翻"这个说法 本身需要被撤回或者软化——这不是说本轮工作没有价值(区间重叠检验和required_n 外推的方法论本身是有价值的产出)，是说"推翻"这个具体措辞可能用早了，需要 和EL.2一样，诚实地说"依然无法判定"。
