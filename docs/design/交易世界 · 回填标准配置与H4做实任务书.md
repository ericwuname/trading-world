# scripts/safe_batch_replace.py

import ast
import re
import shutil
import tempfile
import difflib
from pathlib import Path

class UnsafeReplaceError(Exception):
    pass

def safe_batch_replace(file_paths: list[str], replacements: list[tuple[str, str]],
                         dry_run: bool = True) -> dict:
    """
    对file_paths里的每个文件,依次应用replacements(regex pattern, replacement string)列表,
    先在临时目录的副本上操作,对每个.py文件跑ast.parse验证语法完整。

    dry_run=True(默认): 只报告哪些文件会被改、diff长什么样、语法检查是否通过,不写回真实文件。
    dry_run=False: 语法检查全部通过后,才把临时副本的内容写回真实路径。
    任何一个文件语法检查失败,整个操作全部回滚,不写回任何文件(要么全部成功,要么全部不生效)。
    """
    with tempfile.TemporaryDirectory() as tmpdir:
        tmp_paths = {}
        diffs = {}
        syntax_errors = {}
        for fp in file_paths:
            original = Path(fp).read_text(encoding='utf-8')
            modified = original
            for pattern, repl in replacements:
                modified = re.sub(pattern, repl, modified)
            tmp_path = Path(tmpdir) / Path(fp).name
            tmp_path.write_text(modified, encoding='utf-8')
            tmp_paths[fp] = tmp_path
            diffs[fp] = list(difflib.unified_diff(
                original.splitlines(), modified.splitlines(), lineterm=''))
            if fp.endswith('.py'):
                try:
                    ast.parse(modified)
                except SyntaxError as e:
                    syntax_errors[fp] = str(e)

        if syntax_errors:
            raise UnsafeReplaceError(f"以下文件语法检查失败,操作全部回滚: {syntax_errors}")

        if not dry_run:
            for fp, tmp_path in tmp_paths.items():
                shutil.copy(tmp_path, fp)

        return {"diffs": diffs, "syntax_errors": syntax_errors,
                "applied": not dry_run and not syntax_errors}

import json
from pathlib import Path
from datetime import datetime

REGISTRY_PATH = Path("docs/mutation_registry.json")

def claim_next_ids(n: int, claimed_by: str) -> list[int]:
    """
    读取docs/mutation_registry.json(格式: {"next_available": 51, "history": [...]})，
    原子地分配n个连续编号,更新next_available,并在history里记录
    (编号范围, 领取者标识, 领取时间)。
    以后任何任务书或脚本要新增变异测试,必须先调用这个函数拿编号,
    不允许再靠人工在任务书文字里写死"M45-M49"这种容易过时的编号。
    """
    registry = json.loads(REGISTRY_PATH.read_text())
    start = registry["next_available"]
    ids = list(range(start, start + n))
    registry["next_available"] = start + n
    registry["history"].append({
        "ids": ids, "claimed_by": claimed_by,
        "claimed_at": datetime.now().isoformat()
    })
    REGISTRY_PATH.write_text(json.dumps(registry, indent=2, ensure_ascii=False))
    return ids

from scipy import stats
import numpy as np

def estimate_required_n_for_quadrant_test(observed_diff, observed_pooled_sd,
                                            alpha=0.05, power=0.8, ratio=42/31):
    """
    用观测到的效应量(diff=2.98,来自symmetric_both配置的taker_only vs both_burst)
    和当前的合并标准差,反推达到80%检验功效需要的每组样本量。
    ratio参数保留两组不等长的比例(42:31),不强行假设两组样本量相等。
    用标准的两样本t检验功效公式(可以直接调用statsmodels.stats.power,
    不需要自己重新推导公式)。
    """
    from statsmodels.stats.power import TTestIndPower
    effect_size = observed_diff / observed_pooled_sd  # Cohen's d
    analysis = TTestIndPower()
    n1 = analysis.solve_power(effect_size=effect_size, alpha=alpha, power=power,
                                ratio=ratio, alternative='two-sided')
    return {"effect_size_cohens_d": effect_size, "required_n_group1": n1,
            "required_n_group2": n1 * ratio}
