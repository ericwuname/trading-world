def run_saturation_sweep(configs, shock_sizes=STAGE3_SHOCK_SIZES, seeds=EXISTING_SEEDS):
    """
    对每个config、每档冲击规模,跑paired_impact(复用现有函数,不重写),
    记录fill_ratio,标出从哪一档开始<0.95(饱和边界)。
    对未饱和的档位,记录滑点配对差异(均值、标准误、t值)。
    不调用fit_power_law,这一步只是'看数据长什么样'。
    """
    results = {}
    for name, config in configs.items():
        per_size = []
        for size in shock_sizes:
            paired_diff = measure_paired_slippage(config, size, seeds)  # 复用现有测量函数
            fill_ratio = compute_fill_ratio(config, size, seeds)
            per_size.append({
                "size": size, "fill_ratio": fill_ratio,
                "slippage_mean": paired_diff.mean(), "slippage_sem": paired_diff.sem(),
                "t": paired_diff.mean() / paired_diff.sem() if paired_diff.sem() > 0 else np.nan,
                "saturated": fill_ratio < 0.95
            })
        results[name] = per_size
    return results

# tw/analyzer_concavity.py

def pairwise_local_elasticity(size_small, slippage_small_per_seed,
                                size_big, slippage_big_per_seed):
    """
    对每个种子,分别算局部弹性:
    e_i = log(|slippage_big_i| / |slippage_small_i|) / log(size_big / size_small)
    然后跨种子汇总e_i的均值、标准误、t检验(相对e=1,即线性假设做单样本t检验)。
    这是本工作线的核心函数,必须直接复用现有的滑点测量结果,不重新跑模拟。
    """
    elasticities = np.log(np.abs(slippage_big_per_seed) / np.abs(slippage_small_per_seed)) \
                   / np.log(size_big / size_small)
    t_vs_linear, p_vs_linear = scipy.stats.ttest_1samp(elasticities, 1.0)
    t_vs_sqrt, p_vs_sqrt = scipy.stats.ttest_1samp(elasticities, 0.5)
    return {
        "mean_elasticity": elasticities.mean(), "sem": elasticities.std(ddof=1)/np.sqrt(len(elasticities)),
        "t_vs_linear": t_vs_linear, "p_vs_linear": p_vs_linear,  # 检验"是否显著小于线性"
        "t_vs_sqrt": t_vs_sqrt, "p_vs_sqrt": p_vs_sqrt,          # 检验"是否显著偏离平方根律"
        "n_seeds": len(elasticities)
    }

def concavity_verdict(elasticity_result, alpha=0.05):
    """
    给出一个明确的三分类判决,而不是只报数字:
    - '显著凹'(elasticity的95%置信区间上界 < 1,即显著小于线性)
    - '显著线性或更差'(置信区间下界 >= 1)
    - '不可判定'(置信区间跨过1)
    这个判决就是新的验收标准,取代过去"k是不是接近0.5"这个要求精确拟合的目标。
    """
    ci_upper = elasticity_result["mean_elasticity"] + 1.96 * elasticity_result["sem"]
    ci_lower = elasticity_result["mean_elasticity"] - 1.96 * elasticity_result["sem"]
    if ci_upper < 1.0:
        return "显著凹", (ci_lower, ci_upper)
    elif ci_lower >= 1.0:
        return "显著线性或更差", (ci_lower, ci_upper)
    else:
        return "不可判定", (ci_lower, ci_upper)

# scripts/audit_numeric_claims.py

import re

COMPARISON_PATTERNS = [
    r'从\s*[\d.]+\s*[到→]\s*[\d.]+',
    r'改善了?\s*[\d.]+%',
    r'k\s*[=＝]\s*[\d.]+',
    r'提升|下降|恶化|改善'
]

def scan_report_for_numeric_claims(report_text_path):
    """
    对一期/二期/三线深挖/本轮全部报告文本(已提取的纯文本版本),
    用正则扫描出所有'数值比较型'陈述所在的段落,
    记录每处陈述前后200字符的上下文,供人工/半自动判断
    该处是否伴随了CI、标准误、p值、或至少明确的种子数。
    这一步的产出是'待人工复核清单',不是自动判定的最终结论——
    正则匹配注定有误报漏报,必须留一步人工确认。
    """
    with open(report_text_path, encoding='utf-8') as f:
        text = f.read()
    findings = []
    for pattern in COMPARISON_PATTERNS:
        for match in re.finditer(pattern, text):
            context = text[max(0, match.start()-200):match.end()+200]
            has_ci_nearby = bool(re.search(r'CI|置信区间|标准误|sem|p\s*[=＝]|种子', context))
            findings.append({
                "pattern": pattern, "matched_text": match.group(),
                "context": context, "has_ci_nearby_heuristic": has_ci_nearby
            })
    return findings
