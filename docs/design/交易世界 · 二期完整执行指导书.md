@dataclass
class FundingConfig:
    settle_interval_ticks: int = 8      # 每8个tick结算一次(对应真实8小时周期的刻度)
    funding_sensitivity: float = 0.1    # 溢价->费率的传导系数
    crowding_sensitivity: float = 0.05  # 多头拥挤度->费率的传导系数
    clamp_bp: float = 75.0              # 单次费率上限(bp),防止极端值失控

class PerpetualMarket(Market):
    """
    继承现有Market,不修改撮合逻辑,只在主循环末尾插入结算钩子。
    """
    def __init__(self, *args, funding_config: FundingConfig, fundamental_source, **kwargs):
        super().__init__(*args, **kwargs)
        self.funding_config = funding_config
        self.fundamental_source = fundamental_source  # 复用现有的基本面价值序列,当作"现货代理价"
        self.funding_rate_history = []   # 预分配数组,参考logger.py的写法
        self._last_settle_tick = 0

    def step(self, tick):
        super().step(tick)  # 保持原有撮合流程不变
        if tick - self._last_settle_tick >= self.funding_config.settle_interval_ticks:
            self._settle_funding(tick)
            self._last_settle_tick = tick

    def _compute_premium(self) -> float:
        mid = self.book.get_mid_price()
        anchor = self.fundamental_source.value_at(self.current_tick)
        return (mid - anchor) / anchor

    def _compute_net_long_ratio(self) -> float:
        """
        全市场agent的净多头持仓 / 总持仓绝对值之和。
        必须用当前持仓快照算,不能用历史累计,避免重复计数。
        """
        total_long = sum(max(a.inventory, 0) for a in self.agents)
        total_abs = sum(abs(a.inventory) for a in self.agents)
        return total_long / total_abs if total_abs > 0 else 0.0

    def _settle_funding(self, tick):
        premium = self._compute_premium()
        crowding = self._compute_net_long_ratio()
        raw_rate = (self.funding_config.funding_sensitivity * premium
                    + self.funding_config.crowding_sensitivity * crowding)
        rate = clamp(raw_rate, -self.funding_config.clamp_bp/1e4, self.funding_config.clamp_bp/1e4)

        # 唯一记账路径: 必须调用Market已有的资金变更方法,不能直接改agent.cash
        for agent in self.agents:
            payment = -agent.inventory * rate * self.mark_price_for_notional()
            self._apply_cash_delta(agent, payment, reason="funding_settlement")  # 复用现有记账接口

        self.funding_rate_history.append((tick, rate, premium, crowding))

class FundingArbitrageur(Agent):
    """
    对应真实世界的'现货多头+永续空头收资金费率'策略。
    简化实现:不真的做现货腿(单资产模型里没有现货市场),
    而是用一个'虚拟现货锚定持仓'代表现货腿,只在永续这边下单/收付资金费率,
    现货腿的盈亏按fundamental_source的路径直接记账(不占用订单簿资源)。
    """
    def __init__(self, *args, entry_threshold_bp=5.0, position_size=1.0, **kwargs):
        super().__init__(*args, **kwargs)
        self.entry_threshold_bp = entry_threshold_bp
        self.position_size = position_size
        self.synthetic_spot_pnl = 0.0  # 虚拟现货腿的累计盈亏,单独记录,不进入inventory

    def on_tick(self, market_state):
        recent_rate = market_state.last_funding_rate()
        if recent_rate is None:
            return None
        if recent_rate * 1e4 > self.entry_threshold_bp and self.inventory >= 0:
            return self.sell_order(self.position_size)   # 做空永续收正费率
        elif recent_rate * 1e4 < -self.entry_threshold_bp and self.inventory <= 0:
            return self.buy_order(self.position_size)
        return None

# tw/order_flow/meta_order.py

import numpy as np

@dataclass
class MetaOrderConfig:
    p_meta_start: float = 0.02        # 每tick每个"元订单型agent"启动新元订单的概率
    pareto_alpha: float = 1.5         # 元订单总量的Pareto分布形状参数(重尾)
    pareto_xmin: float = 1.0
    participation_rate: float = 0.15  # 子订单到达强度(越大拆得越快)
    max_child_orders: int = 200       # 防止极端Pareto采样导致无限拆分

class MetaOrderState:
    """挂在agent实例上,跟踪其当前是否处于一个未完成的元订单执行过程中"""
    def __init__(self):
        self.active = False
        self.direction = None      # 'buy' / 'sell'
        self.remaining_qty = 0.0
        self.child_count = 0

class MetaOrderMixin:
    """
    让任意现有Agent子类(如ZeroIntelligence、Chartist)获得元订单行为的Mixin。
    使用方式: class MetaOrderChartist(MetaOrderMixin, Chartist): pass
    """
    def __init__(self, *args, meta_config: MetaOrderConfig, **kwargs):
        super().__init__(*args, **kwargs)
        self.meta_config = meta_config
        self.meta_state = MetaOrderState()
        self._rng_meta = self._spawn_independent_rng("META_ORDER")  # 复用一期"每主体独立随机流"设计

    def maybe_start_meta_order(self, base_direction_signal):
        """
        base_direction_signal: 由子类原有逻辑给出的方向信号(如Chartist的动量方向)。
        只有在没有正在执行的元订单时才可能启动新的。
        """
        if self.meta_state.active:
            return
        if self._rng_meta.random() < self.meta_config.p_meta_start:
            total_qty = self._sample_pareto()
            self.meta_state.active = True
            self.meta_state.direction = base_direction_signal
            self.meta_state.remaining_qty = total_qty
            self.meta_state.child_count = 0

    def _sample_pareto(self):
        u = self._rng_meta.random()
        alpha, xmin = self.meta_config.pareto_alpha, self.meta_config.pareto_xmin
        return xmin * (1 - u) ** (-1 / alpha)

    def get_child_order_qty(self):
        """每次被调用时决定这一tick是否吐出一个子订单,以及吐多少"""
        if not self.meta_state.active:
            return None
        if self._rng_meta.random() > self.meta_config.participation_rate:
            return None  # 这一tick不下子单,保持方向不变,等下次
        child_qty = min(self.meta_state.remaining_qty, self._sample_child_size())
        self.meta_state.remaining_qty -= child_qty
        self.meta_state.child_count += 1
        if (self.meta_state.remaining_qty <= 1e-9
                or self.meta_state.child_count >= self.meta_config.max_child_orders):
            self.meta_state.active = False
        return (self.meta_state.direction, child_qty)

    def _sample_child_size(self):
        return max(0.1, self._rng_meta.expovariate(1.0))

class AdaptiveLiquidityMixin:
    """
    让零智能/做市商在挂单偏移量上,对'最近K笔市场订单的净方向失衡'做出反应。
    这是LMF理论里'流动性提供者预判到可预测订单流,主动稀疏挂单'的实现。
    """
    def __init__(self, *args, flow_window=50, flow_sensitivity=0.3, **kwargs):
        super().__init__(*args, **kwargs)
        self.flow_window = flow_window
        self.flow_sensitivity = flow_sensitivity

    def adjusted_offset(self, base_offset, market_state, side):
        imbalance = market_state.recent_order_flow_imbalance(self.flow_window)
        # imbalance > 0 表示近期买单占优; 若side='sell',应该把卖单价格进一步推高(稀疏化)
        direction_sign = 1 if side == 'sell' else -1
        adaptive_shift = self.flow_sensitivity * imbalance * direction_sign
        return base_offset + max(0.0, adaptive_shift)  # 只允许让挂单更远,不允许倒贴更近

def order_sign_autocorrelation(order_signs: np.ndarray, max_lag: int = 200) -> np.ndarray:
    """
    对逐笔订单方向序列(+1/-1)算自相关函数,返回lag=1..max_lag的ACF值。
    """
    n = len(order_signs)
    x = order_signs - order_signs.mean()
    acf = np.array([np.dot(x[:n-k], x[k:]) / np.dot(x, x) for k in range(1, max_lag+1)])
    return acf

class AdaptiveTrendFollower(Agent):
    """
    候选lookback集合里用epsilon-greedy做bandit学习,
    不是完整的Q-learning(单步决策,没有状态转移),避免过度工程化。
    """
    def __init__(self, *args, candidate_lookbacks=(10, 20, 50, 100),
                 epsilon=0.1, learning_rate=0.05, eval_horizon=20, **kwargs):
        super().__init__(*args, **kwargs)
        self.candidates = list(candidate_lookbacks)
        self.epsilon = epsilon
        self.lr = learning_rate
        self.eval_horizon = eval_horizon
        self.q_values = {lb: 0.0 for lb in self.candidates}
        self.pending_evals = []  # [(tick_chosen, lookback_used, entry_price, direction), ...]
        self._rng_learn = self._spawn_independent_rng("ADAPTIVE_LEARN")

    def on_tick(self, market_state):
        self._resolve_pending_evals(market_state)
        lb = self._choose_lookback()
        momentum = market_state.momentum(lb)
        direction = 1 if momentum > 0 else (-1 if momentum < 0 else 0)
        if direction == 0:
            return None
        self.pending_evals.append({
            "tick_chosen": market_state.tick, "lookback": lb,
            "entry_price": market_state.mid, "direction": direction
        })
        qty = self._position_size()
        return self.buy_order(qty) if direction > 0 else self.sell_order(qty)

    def _choose_lookback(self):
        if self._rng_learn.random() < self.epsilon:
            return self._rng_learn.choice(self.candidates)
        return max(self.q_values, key=self.q_values.get)

    def _resolve_pending_evals(self, market_state):
        still_pending = []
        for ev in self.pending_evals:
            if market_state.tick - ev["tick_chosen"] >= self.eval_horizon:
                realized = ev["direction"] * (market_state.mid - ev["entry_price"]) / ev["entry_price"]
                old_q = self.q_values[ev["lookback"]]
                self.q_values[ev["lookback"]] += self.lr * (realized - old_q)
            else:
                still_pending.append(ev)
        self.pending_evals = still_pending

            avg_pnl = run_result.mean_pnl_per_fixed_momentum_agent()
            stats = run_result.compute_stylized_facts()
            results.append({"adopter_count": count, "seed": seed,
                             "avg_pnl_bp": avg_pnl, **stats})
    return results

class HawkesIntensityProcess:
    """
    单变量自激过程,决定'这个agent群体在下一个极小时间窗口内下单的强度'。
    强度公式: lambda(t) = mu + sum_{t_i < t} alpha * exp(-beta * (t - t_i))
    """
    def __init__(self, mu, alpha, beta):
        assert alpha < beta, "必须满足alpha < beta,否则强度过程不稳定(积分发散)"
        self.mu, self.alpha, self.beta = mu, alpha, beta
        self.event_times = []

    def intensity_at(self, t):
        excitation = sum(self.alpha * np.exp(-self.beta * (t - ti))
                          for ti in self.event_times if ti < t)
        return self.mu + excitation

    def should_fire(self, t, dt, rng):
        """
        用强度做泊松稀疏采样近似:在极小dt内,事件发生概率约为 intensity(t)*dt。
        dt必须远小于1/beta,否则近似失效,需要在配置里加护栏检查。
        """
        p = self.intensity_at(t) * dt
        if rng.random() < p:
            self.event_times.append(t)
            return True
        return False

class CorrelatedFundamentalGenerator:
    """
    生成N个资产的基本面价值序列,带共同因子(市场beta)和特异性因子。
    value_i(t) = value_i(t-1) * (1 + beta_i * common_shock(t) + idio_shock_i(t))
    """
    def __init__(self, n_assets, betas, common_vol, idio_vols, rng):
        self.n_assets = n_assets
        self.betas = betas
        self.common_vol = common_vol
        self.idio_vols = idio_vols
        self.rng = rng
        self.values = np.ones(n_assets) * 100.0  # 初始基准价

    def step(self):
        common_shock = self.rng.gauss(0, self.common_vol)
        for i in range(self.n_assets):
            idio_shock = self.rng.gauss(0, self.idio_vols[i])
            ret = self.betas[i] * common_shock + idio_shock
            self.values[i] *= (1 + ret)
        return self.values.copy()

class MultiAssetMarket:
    """
    管理N个独立的(OrderBook, MatchingEngine)实例,共享CorrelatedFundamentalGenerator。
    每个PerpetualMarket实例(阶段5产出)可以直接复用,只是fundamental_source换成
    CorrelatedFundamentalGenerator对应资产的分量。
    """
    def __init__(self, n_assets, markets: list, fundamental_gen):
        self.n_assets = n_assets
        self.markets = markets
        self.fundamental_gen = fundamental_gen

    def step(self, tick):
        anchors = self.fundamental_gen.step()
        for i, market in enumerate(self.markets):
            market.fundamental_source.set_current(anchors[i])
            market.step(tick)

class PairsTrader(Agent):
    """
    复用本对话真实数据测试里的均值回归逻辑,但这次交易的是'两个资产价格比率的z-score',
    而不是单一资产的价格z-score。
    """
    def __init__(self, *args, asset_a_id, asset_b_id, zwindow=100, zthresh=2.0, **kwargs):
        super().__init__(*args, **kwargs)
        self.asset_a, self.asset_b = asset_a_id, asset_b_id
        self.zwindow, self.zthresh = zwindow, zthresh
        self.ratio_history = deque(maxlen=zwindow)

    def on_tick(self, multi_market_state):
        pa = multi_market_state.mid(self.asset_a)
        pb = multi_market_state.mid(self.asset_b)
        if pa is None or pb is None:
            return None
        ratio = pa / pb
        self.ratio_history.append(ratio)
        if len(self.ratio_history) < self.zwindow:
            return None
        mean, std = np.mean(self.ratio_history), np.std(self.ratio_history)
        if std == 0:
            return None
        z = (ratio - mean) / std
        if z > self.zthresh:
            return {"sell": self.asset_a, "buy": self.asset_b}
        elif z < -self.zthresh:
            return {"buy": self.asset_a, "sell": self.asset_b}
        return None

def simulated_moments_distance(sim_stats: dict, real_stats: dict, weight_matrix=None):
    """
    比阶段4的'距离分数'更严谨: 用协方差矩阵加权(如果条件允许估计的话),
    而不是简单欧氏距离,避免量级不同的指标互相压制(一期已经手工处理了r_ACF(1)分母趋零的问题,
    这里应该系统化地处理所有指标的标准化)。
    """
    moment_names = list(real_stats.keys())
    diff = np.array([sim_stats[k] - real_stats[k] for k in moment_names])
    if weight_matrix is None:
        # 至少用每个指标自己的跨种子标准差做标准化,不能裸用原始差值
        weight_matrix = np.diag([1.0 / max(sim_stats.get(f"{k}_sd", 1.0), 1e-6) for k in moment_names])
    return diff @ weight_matrix @ diff.T

def bootstrap_parameter_uncertainty(optimizer_fn, n_bootstrap=50):
    """
    对真实数据做block bootstrap重采样,每次重新跑一遍校准,
    得到参数估计的分布,而不是单一点估计。
    这是阶段4"训练段/测试段"切分的进阶版:不仅检查过拟合,还量化参数本身的不确定性。
    """
    results = []
    for b in range(n_bootstrap):
        resampled_real_stats = block_bootstrap_resample(real_data, block_size=168)  # 一周为一个block
        best_params = optimizer_fn(resampled_real_stats)
        results.append(best_params)
    return results  # 后续算每个参数的置信区间

这是对本对话开头讨论过的"清算流动性捕获"假设、以及后期讨论的"反身性"和"复杂适应系统" 问题的一次综合仿真——不是为了产出交易信号，是为了看这类连锁反应的传导路径和量级， 给真实世界的风控设计提供参考量级（比如：多大的拥挤度、多高的资产相关性，会让连锁反应 显著放大，这是一个可以指导真实仓位管理的定性结论）。
